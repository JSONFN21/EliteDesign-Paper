\begin{algorithm}[!b]
\caption{EliteDesign}
\label{alg:elite}
\begin{algorithmic}[1]
\Function{Design}{$\vecystar$, \frank, \fweight}
\State $\pcurrent^{(1)}\gets\text{init}(\vecystar)$; \quad $\vecxstar\gets\text{None}$
\For{$t=1,2,\ldots$ until stopped}
    \State $\SAMPLES\gets N$ samples from $(1-\delta)\,\pcurrent^{(t)}+\delta U$
    \State score \SAMPLES with \frank and \fweight
    \State update \vecxstar if any sequence has higher \frank
    \State $\ELITES\gets$ the top 10\% of \SAMPLES by \frank
    \State compute $w_{\mathrm{elite}}$ on \ELITES \Comment{Eq.~\ref{eq:weights}}
    \State update $\pcurrent^{(t+1)}$ toward $p_{\mathrm{elites}}$ \Comment{Eq.~\ref{eq:update}}
    \State score the most probable sequence
    \State update \vecxstar if it has higher \frank
\EndFor
\State rescore the final candidates with ViennaRNA
\State \Return \vecxstar
\EndFunction
\end{algorithmic}
\end{algorithm}
